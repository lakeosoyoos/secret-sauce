# Desktop build dependencies (Windows, Python 3.11 or 3.12).
#
# IMPORTANT: setuptools is pinned. Newer setuptools (71+) makes pkg_resources
# strict about version parsing, which crashes the PACKAGED app at launch with
# "pkg_resources.extern.packaging.version.InvalidVersion: '.../SecretSauce'".
# 65.5.1 keeps the lenient parsing PyInstaller's pkg_resources hook needs.
# (Validated on a Mac test build — same fix applies on Windows.)
setuptools==65.5.1
# pkg_resources (pulled in by Streamlit) imports several VENDORED packages at
# startup — jaraco.*, more-itertools, packaging, platformdirs, appdirs. On a
# clean build machine these exist ONLY inside setuptools' _vendor tree, so
# PyInstaller can miss them and the packaged app dies at launch with
# "the jaraco package is required". Installing them as real top-level packages
# makes them collectable at build time AND gives pkg_resources' extern importer
# a working fallback at runtime. The CI boot self-test is the backstop.
jaraco.text
jaraco.functools
jaraco.context
more-itertools
packaging
platformdirs
appdirs
ordered-set
streamlit
numpy
scipy
matplotlib
openpyxl
pyinstaller
# Build/test-only: pytest runs the desktop test suite (desktop/tests/) in CI
# BEFORE the .exe is built, so an engine/UI regression fails in seconds instead
# of after a multi-minute PyInstaller build. Not imported by the shipped app.
pytest
# Note: weasyprint is intentionally NOT listed — the desktop build is Excel-
# first and excludes it (see SecretSauce.spec). PDF still works if Chrome is
# installed on the machine.
